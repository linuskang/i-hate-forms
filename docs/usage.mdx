# Usage

The `Form` component is designed to stay out of your way. Bring any input, button, or label component and the library handles the wiring.

## Basic text form

```tsx
"use client"

import { Form } from "@/registry/form"
import { Input } from "@/components/ui/input"
import { Button } from "@/components/ui/button"
import { Label } from "@/components/ui/label"

type Values = {
    name: string
    email: string
}

export function BasicForm() {
    return (
        <Form<Values>
            className="space-y-4"
            onSubmit={(values) => alert(JSON.stringify(values))}
        >
            <Form.Title>Subscribe</Form.Title>

            <div className="space-y-2">
                <Label htmlFor="name">Name</Label>
                <Form.Field<Values> name="name">
                    <Input id="name" placeholder="Jane Doe" />
                </Form.Field>
            </div>

            <div className="space-y-2">
                <Label htmlFor="email">Email</Label>
                <Form.Field<Values> name="email">
                    <Input id="email" type="email" placeholder="jane@example.com" />
                </Form.Field>
            </div>

            <Form.Submit>
                <Button>Subscribe</Button>
            </Form.Submit>
        </Form>
    )
}
```

## Validation with Zod

```tsx
import { zodResolver } from "@hookform/resolvers/zod"
import { z } from "zod"

const schema = z.object({
    email: z.string().email("Invalid email"),
    age: z.coerce.number().min(18, "You must be 18+"),
})

type Values = z.infer<typeof schema>

export function ValidatedForm() {
    return (
        <Form<Values>
            formOptions={{ resolver: zodResolver(schema) }}
            onSubmit={(values) => console.log(values)}
        >
            <Form.Field<Values> name="email">
                <Input type="email" />
            </Form.Field>

            <Form.Field<Values> name="age">
                <Input type="number" />
            </Form.Field>

            <Form.Submit>
                <Button>Submit</Button>
            </Form.Submit>
        </Form>
    )
}
```

## Checkboxes and radios

Native inputs are detected automatically.

```tsx
type Values = {
    newsletter: boolean
    plan: "free" | "pro"
}

export function PreferencesForm() {
    return (
        <Form<Values>
            formOptions={{ defaultValues: { newsletter: false, plan: "free" } }}
            onSubmit={(values) => console.log(values)}
        >
            <Form.Field<Values> name="newsletter">
                <input type="checkbox" />
            </Form.Field>

            <Form.Field<Values> name="plan">
                <input type="radio" value="free" /> Free
            </Form.Field>

            <Form.Field<Values> name="plan">
                <input type="radio" value="pro" /> Pro
            </Form.Field>

            <Form.Submit>
                <button type="submit">Save</button>
            </Form.Submit>
        </Form>
    )
}
```

## Custom components

If a component does not expose a standard `value` / `onChange` API, use the `override` prop.

```tsx
import { Slider } from "@/components/ui/slider"

type Values = {
    volume: number[]
}

export function VolumeForm() {
    return (
        <Form<Values>
            formOptions={{ defaultValues: { volume: [50] } }}
            onSubmit={(values) => console.log(values)}
        >
            <Form.Field<Values>
                name="volume"
                override={({ field }) => ({
                    value: field.value,
                    onValueChange: field.onChange,
                })}
            >
                <Slider />
            </Form.Field>

            <Form.Submit>
                <Button>Save</Button>
            </Form.Submit>
        </Form>
    )
}
```

## Showing field errors

Use the render function to access `fieldState`:

```tsx
<Form.Field name="email">
    {({ field, fieldState }) => (
        <div>
            <Input {...field} value={field.value ?? ""} />
            {fieldState.invalid && (
                <p className="text-sm text-red-500">
                    {fieldState.error?.message}
                </p>
            )}
        </div>
    )}
</Form.Field>
```

## Async submission

`Form.Submit` disables the button while `isSubmitting` is true. Combine it with an async `onSubmit` handler.

```tsx
async function onSubmit(values: Values) {
    await fetch("/api/contact", {
        method: "POST",
        body: JSON.stringify(values),
    })
}

<Form<Values> onSubmit={onSubmit}>
    ...
    <Form.Submit>
        <Button>Send</Button>
    </Form.Submit>
</Form>
```
